import React from 'react';
import * as XLSX from 'xlsx';

type Alert = {
  message: string;
};

type ReportData = {
  timestamp: string;
  systemStatus: {
    peopleCount: number;
    riskLevel: string;
    systemStatus: string;
    cameraStatus: string;
    alertsCount: number;
  };
  exportType: string;
  alerts?: Alert[];
};

const sampleReport: ReportData = {
  timestamp: '2025-10-11T18:01:31.798Z',
  systemStatus: {
    peopleCount: 2,
    riskLevel: 'medium',
    systemStatus: 'online',
    cameraStatus: 'active',
    alertsCount: 0,
  },
  exportType: 'incident_report',
  alerts: [
    { message: "High activity detected - 5 people detected" },
    { message: "Motion anomaly detected" },
  ],
};

export function exportReportAsText(report: ReportData) {
  const textContent = `
Report Generated: ${new Date(report.timestamp).toLocaleString()}
People Detected: ${report.systemStatus.peopleCount}
Risk Level: ${report.systemStatus.riskLevel}
System Status: ${report.systemStatus.systemStatus}
Camera Status: ${report.systemStatus.cameraStatus}
Alerts Count: ${report.systemStatus.alertsCount}
Export Type: ${report.exportType}
Alerts:
${report.alerts?.map((a, idx) => `${idx + 1}. ${a.message}`).join('\n') || 'No alerts'}
`;

  const blob = new Blob([textContent], { type: 'text/plain' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = `SafeWatch_Report_${new Date(report.timestamp).toISOString().slice(0,10)}.txt`;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}

export function exportReportAsExcel(report: ReportData) {
  const worksheetData = [
    { Field: 'Timestamp', Value: new Date(report.timestamp).toLocaleString() },
    { Field: 'People Count', Value: report.systemStatus.peopleCount },
    { Field: 'Risk Level', Value: report.systemStatus.riskLevel },
    { Field: 'System Status', Value: report.systemStatus.systemStatus },
    { Field: 'Camera Status', Value: report.systemStatus.cameraStatus },
    { Field: 'Alerts Count', Value: report.systemStatus.alertsCount },
    { Field: 'Export Type', Value: report.exportType },
  ];

  if (report.alerts && report.alerts.length) {
    worksheetData.push({ Field: 'Alerts:', Value: '' });
    report.alerts.forEach((alert, index) => {
      worksheetData.push({ Field: `Alert ${index + 1}`, Value: alert.message });
    });
  }

  const workbook = XLSX.utils.book_new();
  const worksheet = XLSX.utils.json_to_sheet(worksheetData);

  XLSX.utils.book_append_sheet(workbook, worksheet, 'SafeWatch Report');
  XLSX.writeFile(workbook, `SafeWatch_Report_${new Date(report.timestamp).toISOString().slice(0,10)}.xlsx`);
}

const ReportExportButtons: React.FC<{ report: ReportData }> = ({ report }) => {
  return (
    <div>
      <button onClick={() => exportReportAsText(report)}>
        Export Report as Text
      </button>
      <button onClick={() => exportReportAsExcel(report)}>
        Export Report as Excel
      </button>
    </div>
  );
};

export default function App() {
  return (
    <div>
      <h1>SafeWatch Report Export</h1>
      <ReportExportButtons report={sampleReport} />
    </div>
  );
}
